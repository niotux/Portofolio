cmake_minimum_required(VERSION 3.28)

if(UNIX AND NOT APPLE)
  set(CMAKE_C_COMPILER clang-18)
  set(CMAKE_CXX_COMPILER clang++-18)
  set(CMAKE_CXX_COMPILER_CLANG_SCAN_DEPS clang-scan-deps-18)
endif()

project(obsidian)

if(NOT DEFINED CMAKE_BINARY_DIR)
  set(CMAKE_BINARY_DIR ${CMAKE_SOURCE_DIR}/obj)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${OUTPUT_DIR})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${OUTPUT_DIR})
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${OUTPUT_DIR})

foreach(CONFIG ${CMAKE_CONFIGURATION_TYPES})
  string(TOUPPER ${CONFIG} CONFIG_UPPER)
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${CONFIG_UPPER} ${CMAKE_BINARY_DIR}/lib)
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${CONFIG_UPPER} ${CMAKE_BINARY_DIR}/lib)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER} ${CMAKE_BINARY_DIR}/bin)
endforeach()

file(GLOB_RECURSE SOURCES 
  "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.cxx"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.cc"
)

file(GLOB_RECURSE MODULES
  "${CMAKE_CURRENT_SOURCE_DIR}/*.cppm"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.ixx"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.mpp"
)

add_library(${PROJECT_NAME} SHARED ${SOURCES} ${MODULES})

target_include_directories(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})

if(SOURCES)
  target_sources(${PROJECT_NAME} PRIVATE ${SOURCES})
endif()

if(MODULES)
  target_sources(${PROJECT_NAME} 
    PUBLIC FILE_SET CXX_MODULES FILES ${MODULES}
  )
endif()

target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_23)

if(MSVC)
  target_compile_definitions(${PROJECT_NAME} PRIVATE BUILDING_DLL)
endif()

# Include Vulkan configuration
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Vulkan.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/Vulkan.cmake")
  configure_vulkan(${PROJECT_NAME} "${CMAKE_CURRENT_SOURCE_DIR}/config.json")
endif()
